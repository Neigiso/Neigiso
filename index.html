TypeScript import { CharacterStats, SkillConfig, EnemyConfig, AmplifyingReaction, DamageCalculationResult } from '@/types/genshin'; export function calculateDamage( stats: CharacterStats, skill: SkillConfig, enemy: EnemyConfig, reaction: AmplifyingReaction = 'none' ): DamageCalculationResult { // 1. Tính Tổng Tấn Công (Total ATK) const totalBaseAtk = stats.baseAtk; const totalAtk = totalBaseAtk * (1 + stats.atkPercent / 100) + stats.flatAtk; // 2. Base Damage const baseDmg = totalAtk * (skill.multiplier / 100) + (skill.flatBonusDmg || 0); // 3. DMG Bonus Multiplier const dmgBonusMultiplier = 1 + stats.elementalDmgBonus / 100; // 4. Enemy DEF Multiplier const charLevelFactor = stats.level + 100; const enemyDefFactor = (enemy.level + 100) * (1 - enemy.defShredPercent / 100) * (1 - enemy.defIgnorePercent / 100); const defMultiplier = charLevelFactor / (charLevelFactor + enemyDefFactor); // 5. Enemy RES Multiplier const netRes = (enemy.baseResPercent - enemy.resShredPercent) / 100; let resMultiplier = 1; if (netRes < 0) { resMultiplier = 1 - netRes / 2; } else if (netRes <= 0.75) { resMultiplier = 1 - netRes; } else { resMultiplier = 1 / (4 * netRes + 1); } // 6. Amplifying Reaction Multiplier (Melt / Vaporize) let reactionMultiplier = 1; if (reaction !== 'none') { const baseReactionCoeff = reaction.includes('2_0') ? 2.0 : 1.5; // Tăng phần trăm sát thương phản ứng từ Tinh Thông Nguyên Tố (EM) const emBonusPercent = (2.78 * stats.elementalMastery) / (stats.elementalMastery + 1400); reactionMultiplier = baseReactionCoeff * (1 + emBonusPercent); } // 7.
